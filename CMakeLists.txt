cmake_minimum_required(VERSION 3.16)
project(multi_stream_alarm_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wshadow -Wconversion -Wno-unused-parameter)
endif()
find_package(Threads REQUIRED)
# RK3588 真实 HAL（V4L2 采集 + RGA 预处理 + RKNN NPU 推理）编译开关：
#   aarch64/arm64 上默认开启；x86_64 默认关闭（保持纯 mock 可编译可测）。
#   关闭时对应源文件不参与编译，也不链接 rknnrt/rga，跨平台 CI 不受影响。
if(NOT DEFINED MSAD_ENABLE_RKNN)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(MSAD_ENABLE_RKNN_DEFAULT ON)
  else()
    set(MSAD_ENABLE_RKNN_DEFAULT OFF)
  endif()
  set(MSAD_ENABLE_RKNN ${MSAD_ENABLE_RKNN_DEFAULT} CACHE BOOL "Build RK3588 real HAL components (V4L2+RGA+RKNN)")
endif()
set(CORE_SOURCES
  src/common/Logger.cpp src/common/Json.cpp
  src/hal/SimScene.cpp src/hal/MockVideoSource.cpp
  src/hal/MockFrameProcessor.cpp src/hal/MockInferenceEngine.cpp
  src/hal/DurableAlarmNotifier.cpp src/hal/TcpAlarmNotifier.cpp src/algo/Nms.cpp src/algo/PolygonRegion.cpp
  src/algo/PngWriter.cpp src/app/Config.cpp src/app/PreviewPublisher.cpp src/pipeline/VideoPipeline.cpp)
if(MSAD_ENABLE_RKNN)
  find_package(PkgConfig REQUIRED)
  # Proprietary SDK headers/libraries are external dependencies, never vendored.
  set(MSAD_RKNN_ROOT "" CACHE PATH "RKNN SDK prefix containing include/ and lib/")
  find_path(MSAD_RKNN_INCLUDE_DIR rknn_api.h HINTS "${MSAD_RKNN_ROOT}/include")
  find_library(MSAD_RKNN_LIBRARY NAMES rknnrt HINTS "${MSAD_RKNN_ROOT}/lib")
  if(NOT MSAD_RKNN_INCLUDE_DIR OR NOT MSAD_RKNN_LIBRARY)
    message(FATAL_ERROR "External RKNN SDK required: set MSAD_RKNN_ROOT or MSAD_RKNN_INCLUDE_DIR and MSAD_RKNN_LIBRARY")
  endif()
  set(MSAD_MPP_ROOT "" CACHE PATH "Optional private MPP install prefix; never overwrites system libraries")
  if(MSAD_MPP_ROOT)
    # A changed prefix must not silently retain a library from an older SDK.
    unset(MSAD_MPP_LIBRARY CACHE)
    unset(MSAD_MPP_LIBRARY)
    find_library(MSAD_MPP_LIBRARY NAMES rockchip_mpp PATHS "${MSAD_MPP_ROOT}/lib" NO_DEFAULT_PATH)
    if(NOT MSAD_MPP_LIBRARY)
      message(FATAL_ERROR "Private MPP library not found under ${MSAD_MPP_ROOT}/lib")
    endif()
    set(MSAD_MPP_LINK "${MSAD_MPP_LIBRARY}")
    set(MSAD_MPP_INCLUDE "${MSAD_MPP_ROOT}/include/rockchip")
  else()
    pkg_check_modules(MPP REQUIRED IMPORTED_TARGET rockchip_mpp)
    set(MSAD_MPP_LINK PkgConfig::MPP)
    set(MSAD_MPP_INCLUDE /usr/include/rockchip)
  endif()
  pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
  list(APPEND CORE_SOURCES
    src/hal/RealVideoSource.cpp src/hal/RealFrameProcessor.cpp src/hal/RealInferenceEngine.cpp src/hal/RkMppSource.cpp src/hal/DmaBuffer.cpp)
endif()
add_library(msad_core STATIC ${CORE_SOURCES})
target_include_directories(msad_core PUBLIC include)
target_link_libraries(msad_core PUBLIC Threads::Threads)
if(MSAD_ENABLE_RKNN)
  # Board SDK: externally installed RKNN and RGA headers.
  target_compile_definitions(msad_core PUBLIC MSAD_ENABLE_RKNN=1)
  target_include_directories(msad_core SYSTEM PUBLIC ${MSAD_RKNN_INCLUDE_DIR} /usr/include/rga ${MSAD_MPP_INCLUDE})
  target_link_libraries(msad_core PUBLIC ${MSAD_RKNN_LIBRARY} rga ${MSAD_MPP_LINK} PkgConfig::AV)
  message(STATUS "MSAD: RK3588 real HAL enabled (V4L2 + RGA + RKNN)")
endif()
if(WIN32)
  target_link_libraries(msad_core PUBLIC ws2_32)
endif()
add_executable(multi_stream_alarm_demo src/main.cpp)
target_link_libraries(multi_stream_alarm_demo PRIVATE msad_core)
add_executable(multi_stream_alarm_live src/main.cpp)
target_link_libraries(multi_stream_alarm_live PRIVATE msad_core)
add_subdirectory(module/decode_infer)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_subdirectory(module/voice_mq)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(reconnect_tests tests/reconnect_tests.cpp)
  target_link_libraries(reconnect_tests PRIVATE msad_core)
  add_test(NAME reconnect COMMAND reconnect_tests)
  if(MSAD_ENABLE_RKNN)
    add_executable(hardware_tests tests/hardware_tests.cpp)
    target_link_libraries(hardware_tests PRIVATE msad_core)
    # 动态场景检测测评（离线工具）：MPP 解码 → RGA → RKNN 逐帧推理，输出量化指标与带框截图
    add_executable(motion_eval tests/motion_eval.cpp)
    target_link_libraries(motion_eval PRIVATE msad_core)
  endif()
  add_executable(tensor_tests tests/tensor_tests.cpp)
  target_link_libraries(tensor_tests PRIVATE msad_core)
  add_test(NAME tensor COMMAND tensor_tests)
  add_executable(delivery_driver tests/delivery_driver.cpp)
  target_link_libraries(delivery_driver PRIVATE msad_core)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE msad_core)
  add_executable(alarm_tests tests/alarm_tests.cpp)
  target_link_libraries(alarm_tests PRIVATE msad_core)
  add_executable(module_tests tests/module_tests.cpp)
  target_link_libraries(module_tests PRIVATE decode_infer_core)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(voice_tests tests/voice_tests.cpp module/voice_mq/src/VoiceMqSim.cpp)
    target_include_directories(voice_tests PRIVATE module/voice_mq/include)
    target_link_libraries(voice_tests PRIVATE Threads::Threads rt)
  endif()
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    add_test(NAME live_console COMMAND ${Python3_EXECUTABLE} -B -X utf8 -m unittest discover
      -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_live_console.py)
    set_tests_properties(live_console PROPERTIES TIMEOUT 60
      ENVIRONMENT "MSAD_LIVE_BINARY=$<TARGET_FILE:multi_stream_alarm_live>")
    add_test(NAME region_editor COMMAND ${Python3_EXECUTABLE} -B -m unittest discover
      -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_region_editor.py)
    add_test(NAME detection_score COMMAND ${Python3_EXECUTABLE} -B -m unittest discover
      -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_detection_score.py)
    add_test(NAME regression COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tools/check.py --skip-build
      --build-dir $<TARGET_FILE_DIR:multi_stream_alarm_demo>)
    set_tests_properties(regression PROPERTIES TIMEOUT 60)
    add_test(NAME delivery COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_delivery.py $<TARGET_FILE:delivery_driver>)
    set_tests_properties(delivery PROPERTIES TIMEOUT 30)
    add_test(NAME end_to_end COMMAND ${Python3_EXECUTABLE} -X utf8 ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_end_to_end.py $<TARGET_FILE_DIR:multi_stream_alarm_demo>)
    set_tests_properties(end_to_end PROPERTIES TIMEOUT 30)
  endif()
endif()
install(TARGETS multi_stream_alarm_demo RUNTIME DESTINATION bin)
install(FILES config/demo_config.json DESTINATION share/msad)
